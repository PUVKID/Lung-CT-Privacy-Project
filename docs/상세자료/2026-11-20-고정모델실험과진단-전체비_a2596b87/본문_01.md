+    y = np.asarray(labels, dtype=np.int64)

+    p = expit(z) if probabilities is None else np.asarray(probabilities, dtype=np.float64)
+    if not np.isfinite(z).all() or not np.isfinite(p).all() or y.shape != z.shape or y.shape != p.shape:
+        raise ValueError("라벨·로짓·확률의 shape 또는 유한성이 잘못됐습니다.")
+    predicted = (z >= 0).astype(np.int64)
+    matrix = [[int(np.sum((y == a) & (predicted == b))) for b in (0, 1)] for a in (0, 1)]
+    correct = int(np.sum(y == predicted))
+    return {"correct": correct, "n_images": len(y), "accuracy": correct / len(y),
+            "auc": float(roc_auc_score(y, z)), "confusion_matrix": matrix,
+            "loss": float(np.logaddexp(0, -(2 * y - 1) * z).mean()),
+            "logit_min": float(z.min()), "logit_max": float(z.max()),
+            "threshold": 0.5, "logit_threshold": 0.0,
+            "threshold_rounding_differences": int(np.sum(predicted != (p >= 0.5).astype(np.int64)))}
+
+
+def prediction_rows(records, logits, probabilities):
+    if len(records) != len(logits) or len(records) != len(probabilities):
+        raise ValueError("예측 CSV의 원본 목록·값 수가 다릅니다.")
+    return [{"relative_path": row["relative_path"], "label": row["label"], "logit": float(logit),
+             "ng_probability": float(probability), "prediction": int(logit >= 0),
+             "probability_prediction": int(probability >= 0.5)}
+            for row, logit, probability in zip(records, logits, probabilities)]

+def load_model(sigma, device):
+    path = CHECKPOINTS / f"sigma_{sigma}_best.pt"
+    baseline = read_json(BASELINE)
+    known = {item["path"]: item["sha256"] for item in baseline["source_files"]}
+    if str(path) not in known or digest(path) != known[str(path)]:
+        raise ValueError("알려진 최초 SHA의 로컬 체크포인트만 읽을 수 있습니다.")
+    saved = torch.load(path, map_location="cpu", weights_only=True)
+    metrics = read_json(RESULTS / f"sigma_{sigma}_metrics.json")
+    if saved["run_id"] != RUN_ID or saved["sigma"] != sigma or saved["epoch"] != metrics["best_epoch"]:
+        raise ValueError("체크포인트의 실행·강도·선정 epoch가 기존 지표와 다릅니다.")
+    model = SplitResNet50(sigma=sigma, pretrained=False).eval()
+    model.load_state_dict(saved["model"], strict=True)
+    return model.to(device), {"sigma": sigma, "epoch": saved["epoch"], "sha256": known[str(path)], "weights_only": True}

+def infer(model, records, device):
+    dataset = CTDataset(records, image_size=512, training=False)
+    loader = torch.utils.data.DataLoader(dataset, batch_size=8, shuffle=False, num_workers=0)
+    logits, probabilities = [], []
+    with torch.inference_mode():
+        for images, _ in loader:
+            values = model(images.to(device).float()).double()
+            logits.extend(values.cpu().tolist())
+            probabilities.extend(values.sigmoid().cpu().tolist())
+    return logits, probabilities

+def compare_original(sigma, records, logits, probabilities):
+    original = csv_rows(RESULTS / f"sigma_{sigma}_test_predictions_private.csv")
+    if len(original) != len(records):
+        raise AssertionError("원 CSV와 재추론 표본 수가 다릅니다.")
+    if any(a["relative_path"] != b["relative_path"] or int(a["label"]) != b["label"] for a, b in zip(original, records)):
+        raise AssertionError("원 CSV와 재추론 순서·라벨이 다릅니다.")
+    old = np.asarray([float(row["ng_probability"]) for row in original])
+    new = np.asarray(probabilities)
+    prediction_mismatches = int(np.sum((old >= .5) != (np.asarray(logits) >= 0)))
+    result = {"max_probability_abs_difference": float(np.max(np.abs(old - new))),
+              "probabilities_within_prespecified_tolerance": bool(np.allclose(old, new, atol=2e-7, rtol=2e-5)),
+              "prediction_mismatches": prediction_mismatches,
+              "original_correct": int(np.sum((old >= .5) == np.asarray([row["label"] for row in records]))),
+              "original_csv_has_logits": False}
+    if prediction_mismatches or not result["probabilities_within_prespecified_tolerance"]:
+        raise AssertionError("기존 CSV 확률·정답 재현이 사전 허용 범위를 벗어났습니다: " + json.dumps(result))
+    return result

+def cross_sigma(model, records, device, destination):
+    # 각 모듈을 새로 만들어 강도 변경 때 기존 transfer 캐시를 재사용하지 않습니다.
+    blurs = {sigma: GaussianDCTBlur(sigma).to(device) for sigma in SIGMAS}
+    predictions = {sigma: {"logits": [], "probabilities": []} for sigma in SIGMAS}
+    statistics = []
+    dataset = CTDataset(records, image_size=512, training=False)
+    loader = torch.utils.data.DataLoader(dataset, batch_size=8, shuffle=False, num_workers=0)
+    offset = 0
+    with torch.inference_mode():
+        for images, _ in loader:
+            images = images.to(device).float()
+            features = model.edge((images - model.rgb_mean) / model.rgb_std)
+            source64 = features.double()
+            means = source64.mean(dim=(-2, -1), keepdim=True)
+            base_ac = ((source64 - means) ** 2).mean(dim=(1, 2, 3))
+            base_total = (source64 ** 2).mean(dim=(1, 2, 3))
+            for sigma, blur in blurs.items():
+                transmitted = blur(features)
+                logits64 = model.cloud(transmitted).double()
+                predictions[sigma]["logits"].extend(logits64.cpu().tolist())
+                predictions[sigma]["probabilities"].extend(logits64.sigmoid().cpu().tolist())
+                value64 = transmitted.double()
+                new_means = value64.mean(dim=(-2, -1), keepdim=True)
+                ac = ((value64 - new_means) ** 2).mean(dim=(1, 2, 3))
+                mse = ((value64 - source64) ** 2).mean(dim=(1, 2, 3))
+                arrays = {"base_ac_energy": base_ac, "ac_energy": ac,
+                          "ac_retention_ratio": ac / base_ac,
+                          "dc_energy": (new_means ** 2).mean(dim=(1, 2, 3)),
+                          "total_energy": (value64 ** 2).mean(dim=(1, 2, 3)),
+                          "relative_squared_change": mse / base_total,
+                          "mse_from_unblurred": mse,
+                          "max_channel_mean_abs_difference": (new_means - means).abs().amax(dim=(1, 2, 3))}
+                arrays = {name: values.cpu().tolist() for name, values in arrays.items()}
+                for index in range(len(images)):
+                    row = records[offset + index]
+                    statistics.append({"relative_path": row["relative_path"], "label": row["label"], "sigma": sigma,
+                                       **{name: values[index] for name, values in arrays.items()}})
+            offset += len(images)
+            if offset == 8 or offset % 40 == 0 or offset == len(records):
+                progress("cross_sigma", completed_images=offset, total_images=len(records))
+    summary = {}
+    for sigma in SIGMAS:
+        selected = [row for row in statistics if row["sigma"] == sigma]
+        summary[str(sigma)] = {key: float(np.mean([row[key] for row in selected])) for key in selected[0]
+                               if key not in {"relative_path", "label", "sigma"}}
+        summary[str(sigma)]["maximum_channel_mean_error"] = max(row["max_channel_mean_abs_difference"] for row in selected)
+    save_csv(destination / "CT특징_공간통계_비공개.csv", statistics)
+    return predictions, summary
+

+def dc_experiment(model, splits, device, destination):
+    descriptors, labels = {}, {}
+    for name in ["train", "validation", "test"]:
+        loader = torch.utils.data.DataLoader(CTDataset(splits[name], image_size=512, training=False),
+                                             batch_size=8, shuffle=False, num_workers=0)
+        values, targets = [], []
+        with torch.inference_mode():
+            for images, target in loader:
+                features = model.edge((images.to(device).float() - model.rgb_mean) / model.rgb_std)
+                values.append(features.double().mean(dim=(-2, -1)).cpu().numpy())
+                targets.extend(target.int().tolist())
+        descriptors[name] = np.concatenate(values, axis=0)
+        labels[name] = np.asarray(targets, dtype=np.int64)
+        progress("dc_extraction", split=name, n_images=len(targets), n_features=64)
+    scaler = StandardScaler()
+    scaled_train = scaler.fit_transform(descriptors["train"])
+    classifier = LogisticRegression(C=1.0, solver="lbfgs", max_iter=2000, tol=1e-4, random_state=42)
+    # validation/test 정보는 fit·표준화 추정·설정 선택에 사용하지 않습니다.
+    with warnings.catch_warnings(record=True) as captured:
+        warnings.simplefilter("always")
+        classifier.fit(scaled_train, labels["train"])
+    if captured or int(classifier.n_iter_.max()) >= 2000:
+        diagnostic = {"n_iter": classifier.n_iter_.tolist(), "warnings": [str(item.message) for item in captured]}
+        save_json(destination / "DC분류기_수렴진단_비공개.json", diagnostic)
+        raise RuntimeError("DC 분류기의 수렴 경고 또는 반복 상한 도달: " + json.dumps(diagnostic, ensure_ascii=False))
+    reports = {}
+    for name in ["train", "validation", "test"]:
+        transformed = scaler.transform(descriptors[name])
+        logits = classifier.decision_function(transformed)
+        probabilities = classifier.predict_proba(transformed)[:, 1]
+        reports[name] = metric_values(labels[name], logits, probabilities)
+        save_csv(destination / f"DC평균64_{name}_예측_비공개.csv",
+                 prediction_rows(splits[name], logits, probabilities))
+        save_csv(destination / f"DC평균64_{name}_특징_비공개.csv",
+                 [{"relative_path": row["relative_path"], "label": row["label"],
+                   **{f"channel_{channel}": float(value) for channel, value in enumerate(vector)}}
+                  for row, vector in zip(splits[name], descriptors[name])])
+    reports.update(n_features=64, fit_split="train", scaler_fit_split="train", n_fit=769,
+                   parameters={"C": 1.0, "solver": "lbfgs", "max_iter": 2000, "tol": 1e-4, "seed": 42},
+                   n_iter=classifier.n_iter_.tolist(), warnings=[str(item.message) for item in captured],
+                   scaler_mean=scaler.mean_.tolist(), scaler_scale=scaler.scale_.tolist(),
+                   coefficients=classifier.coef_.tolist(), intercept=classifier.intercept_.tolist())
+    return reports

+def open_status_file(path: Path):
+    """원자적 교체를 막지 않는 읽기 전용 바이너리 파일을 엽니다."""
+    if os.name != "nt":
+        return path.open("rb")
+    import msvcrt
+    kernel = ctypes.WinDLL("kernel32", use_last_error=True)
+    kernel.CreateFileW.argtypes = [ctypes.c_wchar_p, ctypes.c_uint32, ctypes.c_uint32,
+                                  ctypes.c_void_p, ctypes.c_uint32, ctypes.c_uint32,
+                                  ctypes.c_void_p]
+    kernel.CreateFileW.restype = ctypes.c_void_p
+    kernel.CloseHandle.argtypes = [ctypes.c_void_p]
+    kernel.CloseHandle.restype = ctypes.c_int
+    # 삭제 공유는 쓰기 권한을 부여하지 않으며 읽는 동안 이름 교체를 허용합니다.
+    for delay in (0.01, 0.02, 0.04, 0.08, None):
+        handle = kernel.CreateFileW(str(path), 0x80000000, 0x00000007, None, 3, 0x80, None)
+        if handle != ctypes.c_void_p(-1).value:
+            break
+        error_code = ctypes.get_last_error()
+        # 이름 교체 순간의 파일/경로 부재와 공유 위반만 짧게 재시도합니다.
+        if error_code not in (2, 3, 32) or delay is None:
+            raise ctypes.WinError(error_code)
+        time.sleep(delay)
+    try:
+        descriptor = msvcrt.open_osfhandle(handle, os.O_RDONLY | os.O_BINARY)
+    except BaseException:
+        kernel.CloseHandle(handle)
+        raise
+    try:
+        # 여기부터 파일 객체가 descriptor와 원래 Windows 핸들의 닫기를 소유합니다.
+        return os.fdopen(descriptor, "rb")
+    except BaseException:
+        os.close(descriptor)
+        raise

+    squares = (descriptors * descriptors).sum(axis=1)
+    distance_squared = np.maximum(squares[:, None] + squares[None, :] - 2 * descriptors @ descriptors.T, 0)
+    connected = np.triu(distance_squared <= float(similarity_rmse) ** 2 * descriptors.shape[1], 1)
+    for left, right in zip(*np.nonzero(connected)):
+        parents[find(int(left))] = find(int(right))
+    members = defaultdict(list)
+    for index in range(len(rows)):
+        members[find(index)].append(index)
+    for indices in members.values():
+        if len({rows[i]["label"] for i in indices}) > 1:
+            raise ValueError("완전 중복 또는 근접 유사 그룹에 OK와 NG 라벨 충돌이 있습니다. 원본 확인이 필요합니다.")
+        group = min(rows[i]["pixel_sha256"] for i in indices)
+        for index in indices:
+            rows[index]["duplicate_group"] = group
+            rows[index]["similarity_group"] = group
+    return rows
+
+
+def split_records(records, seed=42, validation_ratio=0.15, test_ratio=0.15):
+    if not (0 < validation_ratio < 1 and 0 < test_ratio < 1 and validation_ratio + test_ratio < 1):
+        raise ValueError("학습·검증·시험 비율이 유효하지 않습니다.")
+    by_class = defaultdict(dict)
+    for row in records:
+        by_class[row["label"]].setdefault(row["duplicate_group"], []).append(row)
+    if set(by_class) != {0, 1}:
+        raise ValueError("OK와 NG 두 클래스가 모두 필요합니다.")
+    result = {"train": [], "validation": [], "test": []}
+    ratios = {"train": 1 - validation_ratio - test_ratio, "validation": validation_ratio, "test": test_ratio}
+    rng = np.random.default_rng(seed)
+    for label in [0, 1]:
+        groups = sorted(by_class[label])
+        if len(groups) < 7:
+            raise ValueError("각 클래스에 서로 다른 영상 중복 그룹이 7개 이상 필요합니다.")
+        groups = [groups[i] for i in rng.permutation(len(groups))]
+        groups.sort(key=lambda group: len(by_class[label][group]), reverse=True)
+        totals = Counter(row["source_class"] for group in groups for row in by_class[label][group])
+        allocated = {name: Counter() for name in result}
+        for group in groups:
+            members = by_class[label][group]
+            counts = Counter(row["source_class"] for row in members)
+            # 혼합 원본 클래스 그룹도 쪼개지 않고 각 원본 클래스의 목표 비율을 함께 반영합니다.
+            scores = {name: sum(count * (allocated[name][source] + count / 2) /
+                               max(totals[source] * ratios[name], 1) for source, count in counts.items()) / len(members)
+                      for name in result}
+            chosen = min(scores, key=scores.get)
+            result[chosen].extend(members)
+            allocated[chosen].update(counts)
+    for name in result:
+        result[name].sort(key=lambda row: row["relative_path"])
+    assert_disjoint(result)
+    if any({row["label"] for row in rows} != {0, 1} for rows in result.values()):
+        raise ValueError("유사 그룹과 원본 클래스 분포로 세 분할에 두 라벨을 모두 배정하지 못했습니다.")
+    return result
+
+
+def assert_disjoint(splits):
+    keys = ["path", "byte_sha256", "pixel_sha256", "duplicate_group"]
+    if all("similarity_group" in row for rows in splits.values() for row in rows):
+        keys += ["similarity_group", "exact_duplicate_group"]
+    for key in keys:
+        owners = {}
+        for name, rows in splits.items():
+            for row in rows:
+                previous = owners.setdefault(row[key], name)
+                if previous != name:
+                    raise ValueError(f"분할 사이 {key} 누수가 있습니다.")
+
+
+def balanced_epoch_plan(records, samples_per_class, seed, epoch):
+    if isinstance(samples_per_class, bool) or int(samples_per_class) != samples_per_class or samples_per_class < 1:
+        raise ValueError("클래스별 표본 수는 양의 정수여야 합니다.")
+    rng = np.random.default_rng(np.random.SeedSequence([int(seed), int(epoch)]))
+    plan = []
+    for label in [0, 1]:
+        indices = [i for i, row in enumerate(records) if row["label"] == label]
+        if not indices:
+            raise ValueError("학습에 OK와 NG가 모두 필요합니다.")
+        selected = []
+        while len(selected) < samples_per_class:
+            selected.extend(rng.permutation(indices).tolist())
+        for index in selected[:samples_per_class]:
+            plan.append((int(index), int(rng.integers(0, 2 ** 63 - 1))))
+    return [plan[i] for i in rng.permutation(len(plan))]
+
+
+class CTDataset(Dataset):
+    def __init__(self, records, image_size=512, training=False, samples_per_class=5000, seed=42):
+        self.records = list(records)
+        self.image_size = image_size
+        self.training = training
+        self.samples_per_class = samples_per_class
+        self.seed = seed
+        self.cache = {}
+        self.set_epoch(0)
+
+    def set_epoch(self, epoch):
+        self.plan = balanced_epoch_plan(self.records, self.samples_per_class, self.seed, epoch) if self.training else [(i, 0) for i in range(len(self.records))]
+
+    def __len__(self):
+        return len(self.plan)
+
+    def __getitem__(self, item):
+        index, sample_seed = self.plan[item]
+        row = self.records[index]
+        path = row["path"]
+        if path not in self.cache:
+            with Image.open(path) as image:
+                gray = image.convert("L")
+                if gray.size != (self.image_size, self.image_size):
+                    gray = gray.resize((self.image_size, self.image_size), Image.Resampling.BILINEAR)
+                self.cache[path] = np.asarray(gray, dtype=np.uint8).copy()
+        pixels = self.cache[path]
+        if self.training:
+            rng = np.random.default_rng(sample_seed)
+            image = Image.fromarray(pixels)
+            if rng.random() < 0.5:
+                image = image.transpose(Image.Transpose.FLIP_LEFT_RIGHT)
+            image = ImageEnhance.Contrast(image).enhance(float(rng.uniform(0.94, 1.06)))
+            pixels = np.clip(np.asarray(image, dtype=np.float32) + rng.uniform(-6, 6), 0, 255)
+        tensor = torch.from_numpy(np.asarray(pixels, dtype=np.float32).copy()).unsqueeze(0).repeat(3, 1, 1) / 255.0
+        return tensor, torch.tensor(float(row["label"]), dtype=torch.float32)
+
+
+def dataset_summary(records, splits, samples_per_class):
+    def count(rows):
+        labels = Counter(row["label"] for row in rows)
+        return {"OK": labels[0], "NG": labels[1], "total": len(rows)}
+    return {"source_counts": dict(Counter(row["source_class"] for row in records)),
+            "class_counts": count(records), "splits": {name: count(rows) for name, rows in splits.items()},
+            "original_total": len(records), "unique_duplicate_groups": len({r["exact_duplicate_group"] for r in records}),
+            "duplicate_extra_files": len(records) - len({r["exact_duplicate_group"] for r in records}),
+            "similarity_group_count": len({r["similarity_group"] for r in records}),
+            "largest_similarity_group": max(Counter(r["similarity_group"] for r in records).values()),
+            "source_counts_by_split": {name: dict(Counter(row["source_class"] for row in rows)) for name, rows in splits.items()},
+            "epoch_target_per_class": samples_per_class, "split_unit": "image_similarity_group",
+            "patient_id_available": False, "leakage_checked": True,
+            "limitations": ["실제 환자 ID가 없어 이미지 단위 예비실험입니다. 환자 간 독립성을 검증하지 못했습니다.",
+                            "완전 중복과 고정 64픽셀 RMSE 연결 성분을 묶지만 이 그룹은 환자 ID가 아닙니다. 같은 환자의 다른 영상은 남을 수 있습니다.",
+                            "온라인 증강 5000개/클래스는 원본 환자나 독립 영상 5000개를 의미하지 않습니다."]}
+
+
+def validate_reference_manifest(config, root, payload):
+    """기준 원본 목록과 실제 분할을 고정 해시로 대조하고 불일치하면 저장하지 않습니다."""
+    keys = ("expected_reference_manifest", "expected_reference_manifest_sha256", "expected_reference_records_sha256")
+    if not any(key in config for key in keys):
+        return
+    if any(not isinstance(config.get(key), str) or not config[key] for key in keys):
+        raise ValueError("기준 manifest 경로·파일 해시·원본 목록 해시가 모두 필요합니다.")
+    for key in keys[1:]:
+        if len(config[key]) != 64 or any(c not in "0123456789abcdef" for c in config[key]):
+            raise ValueError("기준 manifest 해시 형식이 유효하지 않습니다.")
+    path = Path(config[keys[0]])
+    if not path.is_absolute():
+        path = Path(root) / path
+    if path.is_symlink():
+        raise ValueError("기준 manifest는 폴더 연결을 따라 읽지 않습니다.")
+    raw = path.read_bytes()
+    if hashlib.sha256(raw).hexdigest() != config[keys[1]]:
+        raise ValueError("기준 manifest 파일 해시가 고정값과 다릅니다.")
+    reference = json.loads(raw.decode("utf-8"))
+    if not isinstance(reference, dict):
+        raise ValueError("기준 manifest 형식이 유효하지 않습니다.")
+    reference_digest = hashlib.sha256(json.dumps(reference.get("records"), ensure_ascii=False, sort_keys=True).encode("utf-8")).hexdigest()
+    payload_digest = hashlib.sha256(json.dumps(payload.get("records"), ensure_ascii=False, sort_keys=True).encode("utf-8")).hexdigest()
+    if (reference_digest != config[keys[2]] or payload_digest != config[keys[2]]
+            or reference.get("manifest_sha256") != reference_digest
+            or payload.get("manifest_sha256") != payload_digest
+            or payload.get("records") != reference.get("records")
+            or payload.get("splits") != reference.get("splits")):
+        raise ValueError("기준 manifest의 원본 목록 또는 정확한 분할과 다릅니다.")
+
+
+def prepare_dataset(config, root):
+    records = discover_images(config["dataset_root"], config["similarity_rmse_threshold"], config["similarity_image_size"])
+    actual = dict(Counter(row["source_class"] for row in records))
+    expected = config.get("expected_source_counts")
+    if expected and actual != expected:
+        raise ValueError(f"원본 클래스별 수가 확인된 명세와 다릅니다: 실제={actual}, 예상={expected}")
+    splits = split_records(records, config["seed"], config["validation_ratio"], config["test_ratio"])
+    summary = dataset_summary(records, splits, config["samples_per_class"])
+    payload = {"records": records, "splits": splits, "summary": summary,
+               "grouping_policy": {"image_size": config["similarity_image_size"], "rmse_threshold_8bit": config["similarity_rmse_threshold"],
+                                   "algorithm": "exact-union-plus-label-independent-rmse-connected-components-float64-v1"},
+               "manifest_sha256": hashlib.sha256(json.dumps(records, ensure_ascii=False, sort_keys=True).encode("utf-8")).hexdigest()}
+    validate_reference_manifest(config, root, payload)
+    destination = Path(root) / "work/results"
+    destination.mkdir(parents=True, exist_ok=True)
+    path = destination / "dataset_manifest_private.json"
+    path.write_text(json.dumps(payload, ensure_ascii=False, indent=2, allow_nan=False) + "\n", encoding="utf-8")
+    return payload
