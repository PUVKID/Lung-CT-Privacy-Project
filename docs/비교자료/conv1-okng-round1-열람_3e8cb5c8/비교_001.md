+def process_alive(pid):
+    """PID가 있을 때만 운영체제에서 생존 여부를 확인합니다."""
+    if not isinstance(pid, int) or isinstance(pid, bool) or pid <= 0:
+        return None
+    if os.name == "nt":
+        kernel = ctypes.WinDLL("kernel32", use_last_error=True)
+        kernel.OpenProcess.restype = ctypes.c_void_p
+        handle = kernel.OpenProcess(0x1000, False, pid)
+        if not handle:
+            return None if ctypes.get_last_error() == 5 else False
+        try:
+            code = ctypes.c_ulong()
+            if not kernel.GetExitCodeProcess(ctypes.c_void_p(handle), ctypes.byref(code)):
+                return None
+            return code.value == 259
+        finally:
+            kernel.CloseHandle(ctypes.c_void_p(handle))
+    try:
+        os.kill(pid, 0)
+        return True
+    except ProcessLookupError:
+        return False
+    except PermissionError:
+        return None

+def open_status_file(path: Path):
+    """원자적 교체를 막지 않는 읽기 전용 바이너리 파일을 엽니다."""
+    if os.name != "nt":
+        return path.open("rb")
+    import msvcrt
+    kernel = ctypes.WinDLL("kernel32", use_last_error=True)
+    kernel.CreateFileW.argtypes = [ctypes.c_wchar_p, ctypes.c_uint32, ctypes.c_uint32,
+                                  ctypes.c_void_p, ctypes.c_uint32, ctypes.c_uint32,
+                                  ctypes.c_void_p]
+    kernel.CreateFileW.restype = ctypes.c_void_p
+    kernel.CloseHandle.argtypes = [ctypes.c_void_p]
+    kernel.CloseHandle.restype = ctypes.c_int
+    # 삭제 공유는 쓰기 권한을 부여하지 않으며 읽는 동안 이름 교체를 허용합니다.
+    for delay in (0.01, 0.02, 0.04, 0.08, None):
+        handle = kernel.CreateFileW(str(path), 0x80000000, 0x00000007, None, 3, 0x80, None)
+        if handle != ctypes.c_void_p(-1).value:
+            break
+        error_code = ctypes.get_last_error()
+        # 이름 교체 순간의 파일/경로 부재와 공유 위반만 짧게 재시도합니다.
+        if error_code not in (2, 3, 32) or delay is None:
+            raise ctypes.WinError(error_code)
+        time.sleep(delay)
+    try:
+        descriptor = msvcrt.open_osfhandle(handle, os.O_RDONLY | os.O_BINARY)
+    except BaseException:
+        kernel.CloseHandle(handle)
+        raise
+    try:
+        # 여기부터 파일 객체가 descriptor와 원래 Windows 핸들의 닫기를 소유합니다.
+        return os.fdopen(descriptor, "rb")
+    except BaseException:
+        os.close(descriptor)
+        raise

+def read_status(root: Path):
+    meta = {"observed_at": datetime.now(timezone.utc).isoformat(), "stale": True,
+            "age_seconds": None, "process_alive": None, "read_error": None}
+    try:
+        path = root / "work" / "runtime" / "status.json"
+        # 상태 파일이 다른 위치로 연결되면 읽지 않습니다.
+        if path.is_symlink() or not path.resolve().is_relative_to(root.resolve()):
+            raise ValueError("상태 파일 경계 위반")
+        with open_status_file(path) as stream:
+            payload = stream.read(MAX_STATUS_BYTES + 1)
+        if len(payload) > MAX_STATUS_BYTES:
+            raise ValueError("상태 파일이 허용 크기를 초과했습니다")
+        data = json.loads(payload.decode("utf-8"), parse_constant=reject_nonfinite)

+def make_server(root=ROOT, port=8766, token=None):
+    root = Path(root).resolve()
+    session_token = token or secrets.token_urlsafe(32)
+
+    class Handler(BaseHTTPRequestHandler):
+        def log_message(self, *_args):
+            pass
+
+        def send_content(self, status, body, content_type="text/plain; charset=utf-8", headers=None):
+            self.send_response(status)
+            self.send_header("Content-Type", content_type)
+            self.send_header("Content-Length", str(len(body)))
+            self.send_header("Cache-Control", "no-store")
+            self.send_header("X-Content-Type-Options", "nosniff")
+            self.send_header("Referrer-Policy", "no-referrer")

+            for key, value in (headers or {}).items():
+                self.send_header(key, value)
+            self.end_headers()
+            self.wfile.write(body)

+        def do_GET(self):
+            # DNS 재바인딩 및 다른 사이트의 교차 출처 조회를 차단합니다.
+            expected_host = f"127.0.0.1:{self.server.server_port}"

+                self.send_content(HTTPStatus.FORBIDDEN, "허용되지 않은 호스트".encode())
+                return
+            parts = urlsplit(self.path)
+            supplied = parse_qs(parts.query).get("token", [""])[0]
+            if parts.path == "/" and supplied and secrets.compare_digest(supplied, session_token):
+                self.send_content(HTTPStatus.SEE_OTHER, b"", headers={

+                return
+            cookies = SimpleCookie()
+            try:
+                cookies.load(self.headers.get("Cookie", ""))
+                value = cookies["dashboard_session"].value if "dashboard_session" in cookies else ""
+            except Exception:
+                value = ""
+            if not secrets.compare_digest(value, session_token):
+                self.send_content(HTTPStatus.UNAUTHORIZED, "세션 URL로 접속해 주세요".encode())
+                return
+            if parts.path == "/api/status" and not parts.query:
+                payload = json.dumps(read_status(root), ensure_ascii=False, allow_nan=False).encode("utf-8")
+                self.send_content(HTTPStatus.OK, payload, "application/json; charset=utf-8")
+                return
+            if parts.path not in ASSETS or parts.query:
+                self.send_content(HTTPStatus.NOT_FOUND, "등록되지 않은 경로".encode())
+                return
+            web = (root / "web").resolve()
+            path = web / ASSETS[parts.path]
+            if not web.is_relative_to(root) or path.is_symlink() or not path.resolve().is_relative_to(web):
+                self.send_content(HTTPStatus.FORBIDDEN, b"Forbidden")
+                return
+            try:
+                content = path.read_bytes()
+            except OSError:
+                self.send_content(HTTPStatus.NOT_FOUND, b"Not found")
+                return
+            mime = "text/javascript" if path.suffix == ".js" else mimetypes.guess_type(path.name)[0] or "application/octet-stream"
+            self.send_content(HTTPStatus.OK, content, mime)
+

+# 실행 방법
+
+PowerShell에서 프로젝트 1차 폴더를 작업 디렉터리로 열고 전용 환경을 사용합니다. 이미 실행 중인 학습이 있는지 먼저 확인하세요. 각 명령은 실제 상태 파일을 갱신하므로 진행 중에는 준비나 연결 검사도 중복 실행하지 마세요.
+
+```powershell
+.\.venv\Scripts\python.exe run_training.py prepare
+.\.venv\Scripts\python.exe run_training.py smoke --device cuda
+.\.venv\Scripts\python.exe run_training.py train --device cuda
+```
+

+"""원본을 바꾸지 않고 중복 단위 분할과 학습 전용 증강을 수행합니다."""

+import json

+IMAGE_SUFFIXES = {".png", ".jpg", ".jpeg", ".bmp", ".tif", ".tiff"}

+def binary_label(value):
+    key = str(value).lower().strip().replace(" ", "").replace("_", "").replace("-", "")
+    if key in {"0", "ok", "bengin", "bengincases", "benign", "benigncases", "normal", "normalcases"}:
+        return 0
+    if key in {"1", "ng", "malignant", "malignantcases"}:
+        return 1
+    raise ValueError(f"확인되지 않은 원본 라벨: {value!r}")

+def discover_images(dataset_root):
+    root = Path(dataset_root).resolve()
+    if not root.is_dir():
+        raise FileNotFoundError(f"원본 자료 폴더가 없습니다: {root}")
+    rows = []
+    for folder in sorted(root.iterdir()):
+        if not folder.is_dir():
+            continue
+        try:
+            label = binary_label(folder.name)
+        except ValueError:
+            continue
+        for path in sorted(folder.rglob("*")):
+            if not path.is_file() or path.suffix.lower() not in IMAGE_SUFFIXES:
+                continue
+            raw = path.read_bytes()
+            with Image.open(path) as image:
+                image.load()
+                gray = image.convert("L")
+                width, height = gray.size
+                # 실제 학습에 읽히는 회색조 픽셀과 크기를 함께 해시합니다.
+                pixel_digest = hashlib.sha256(f"L:{width}:{height}:".encode() + gray.tobytes()).hexdigest()
+            rows.append({"path": str(path.resolve()), "relative_path": path.relative_to(root).as_posix(),
+                         "source_class": folder.name, "label": label,
+                         "byte_sha256": hashlib.sha256(raw).hexdigest(), "pixel_sha256": pixel_digest,
+                         "width": width, "height": height})
+    if not rows:
+        raise ValueError("분류 가능한 원본 영상이 0개입니다.")
+    parents = list(range(len(rows)))
+
+    def find(index):
+        while parents[index] != index:
+            parents[index] = parents[parents[index]]
+            index = parents[index]
+        return index
+
+    known = {}
+    for index, row in enumerate(rows):
+        for kind in ["byte_sha256", "pixel_sha256"]:
+            key = (kind, row[kind])
+            if key in known:
+                parents[find(index)] = find(known[key])
+            else:
+                known[key] = index
+    members = defaultdict(list)
+    for index in range(len(rows)):
+        members[find(index)].append(index)
+    for indices in members.values():
+        if len({rows[i]["label"] for i in indices}) > 1:
+            raise ValueError("동일 바이트/픽셀 영상에 OK와 NG 라벨 충돌이 있습니다. 원본 확인이 필요합니다.")
+        group = min(rows[i]["pixel_sha256"] for i in indices)
+        for index in indices:
+            rows[index]["duplicate_group"] = group
+    return rows

+def split_records(records, seed=42, validation_ratio=0.15, test_ratio=0.15):
+    if not (0 < validation_ratio < 1 and 0 < test_ratio < 1 and validation_ratio + test_ratio < 1):
+        raise ValueError("학습·검증·시험 비율이 유효하지 않습니다.")
+    by_class = defaultdict(dict)
+    for row in records:
+        by_class[row["label"]].setdefault(row["duplicate_group"], []).append(row)
+    if set(by_class) != {0, 1}:
+        raise ValueError("OK와 NG 두 클래스가 모두 필요합니다.")
+    result = {"train": [], "validation": [], "test": []}
+    rng = np.random.default_rng(seed)
+    for label in [0, 1]:
+        groups = sorted(by_class[label])
+        if len(groups) < 7:
+            raise ValueError("각 클래스에 서로 다른 영상 중복 그룹이 7개 이상 필요합니다.")
+        groups = [groups[i] for i in rng.permutation(len(groups))]
+        n_test = max(1, round(len(groups) * test_ratio))
+        n_val = max(1, round(len(groups) * validation_ratio))
+        n_train = len(groups) - n_test - n_val
+        if n_train < 1:
+            raise ValueError("원본 학습 그룹이 남지 않는 분할입니다.")
+        for name, selected in [("test", groups[:n_test]), ("validation", groups[n_test:n_test + n_val]),
+                               ("train", groups[n_test + n_val:])]:
+            for group in selected:
+                result[name].extend(by_class[label][group])
+    for name in result:
+        result[name].sort(key=lambda row: row["relative_path"])
+    assert_disjoint(result)
+    return result
+
+
+def assert_disjoint(splits):
+    for key in ["path", "byte_sha256", "pixel_sha256", "duplicate_group"]:
+        owners = {}
+        for name, rows in splits.items():
+            for row in rows:
+                previous = owners.setdefault(row[key], name)
+                if previous != name:
+                    raise ValueError(f"분할 사이 {key} 누수가 있습니다.")
+
+
+def balanced_epoch_plan(records, samples_per_class, seed, epoch):
+    if isinstance(samples_per_class, bool) or int(samples_per_class) != samples_per_class or samples_per_class < 1:
+        raise ValueError("클래스별 표본 수는 양의 정수여야 합니다.")
+    rng = np.random.default_rng(np.random.SeedSequence([int(seed), int(epoch)]))
+    plan = []
+    for label in [0, 1]:
+        indices = [i for i, row in enumerate(records) if row["label"] == label]
+        if not indices:
+            raise ValueError("학습에 OK와 NG가 모두 필요합니다.")
+        selected = []
+        while len(selected) < samples_per_class:
+            selected.extend(rng.permutation(indices).tolist())
+        for index in selected[:samples_per_class]:
+            plan.append((int(index), int(rng.integers(0, 2 ** 63 - 1))))
+    return [plan[i] for i in rng.permutation(len(plan))]
+

+def _dct(x):
+    n = x.shape[-1]
+    extension = torch.cat((x, x.flip(-1)), dim=-1)
+    spectrum = torch.fft.rfft(extension, dim=-1)[..., :n]
+    angle = torch.arange(n, device=x.device, dtype=x.dtype) * (-math.pi / (2 * n))
+    raw = (spectrum * torch.polar(torch.ones_like(angle), angle)).real
+    scale = torch.full((n,), math.sqrt(1 / (2 * n)), device=x.device, dtype=x.dtype)
+    scale[0] = math.sqrt(1 / (4 * n))
+    return raw * scale
+
+
+def _idct(x):
+    n = x.shape[-1]
+    scale = torch.full((n,), math.sqrt(2 * n), device=x.device, dtype=x.dtype)
+    scale[0] = math.sqrt(4 * n)
+    angle = torch.arange(n, device=x.device, dtype=x.dtype) * (math.pi / (2 * n))
+    spectrum = (x * scale) * torch.polar(torch.ones_like(angle), angle)
+    spectrum = torch.cat((spectrum, torch.zeros_like(spectrum[..., :1])), dim=-1)
+    return torch.fft.irfft(spectrum, n=2 * n, dim=-1)[..., :n]
+
+
+class GaussianDCTBlur(nn.Module):
+    def __init__(self, sigma):
+        super().__init__()
+        if not math.isfinite(float(sigma)) or sigma < 0:
+            raise ValueError("sigma는 유한한 0 이상이어야 합니다.")
+        self.sigma = float(sigma)
+        self.register_buffer("transfer", torch.empty(0), persistent=False)
+
+    def forward(self, inputs):
+        if inputs.ndim != 4 or min(inputs.shape[-2:]) < 2:
+            raise ValueError("블러 입력은 N×C×H×W이며 H,W는 2 이상이어야 합니다.")
+        if self.sigma == 0:
+            return inputs
+        original_dtype = inputs.dtype
+        x = inputs.float()
+        height, width = x.shape[-2:]
+        if self.transfer.shape != (height, width) or self.transfer.device != x.device:
+            yy = torch.arange(height, device=x.device, dtype=torch.float32) * (math.pi / height)
+            xx = torch.arange(width, device=x.device, dtype=torch.float32) * (math.pi / width)
+            self.transfer = torch.exp(-0.5 * self.sigma ** 2 * (yy[:, None].square() + xx[None, :].square()))
+        # DCT-II의 반사/Neumann 경계를 사용하며 sigma는 특징맵 픽셀 단위입니다.
+        coefficients = _dct(_dct(x).transpose(-1, -2)).transpose(-1, -2)
+        coefficients = coefficients * self.transfer
+        result = _idct(_idct(coefficients).transpose(-1, -2)).transpose(-1, -2)
+        return result.to(original_dtype)
+
+
+class CloudResNet(nn.Module):
+    def __init__(self, backbone):
+        super().__init__()
+        for name in ["bn1", "relu", "maxpool", "layer1", "layer2", "layer3", "layer4", "avgpool"]:
+            setattr(self, name, getattr(backbone, name))
+        self.head = nn.Sequential(nn.Linear(2048, 128), nn.ReLU(), nn.Dropout(0.3), nn.Linear(128, 1))
+        for parameter in self.parameters():
+            parameter.requires_grad_(False)
+        for parameter in self.layer4.parameters():
+            parameter.requires_grad_(True)
+        for module in self.modules():
+            if isinstance(module, nn.BatchNorm2d):
+                for parameter in module.parameters():
+                    parameter.requires_grad_(False)
+        for parameter in self.head.parameters():
+            parameter.requires_grad_(True)
+
+    def forward(self, x):
+        # 고정된 앞부분에는 역전파 그래프가 필요하지 않습니다.
+        with torch.no_grad():
+            x = self.maxpool(self.relu(self.bn1(x)))
+            x = self.layer3(self.layer2(self.layer1(x)))
+        x = self.layer4(x)
+        x = self.avgpool(x).flatten(1)
+        return self.head(x).squeeze(1)
+
+
+class SplitResNet50(nn.Module):
+    def __init__(self, sigma=0, pretrained=True):
+        super().__init__()
+        # 가중치가 없으면 실제 학습을 임의 난수 초기값으로 대체하지 않습니다.
+        backbone = resnet50(weights=ResNet50_Weights.IMAGENET1K_V1 if pretrained else None, progress=True)
+        self.edge = backbone.conv1
+        self.edge.requires_grad_(False)
+        self.blur = GaussianDCTBlur(sigma)
+        self.cloud = CloudResNet(backbone)
+        self.register_buffer("rgb_mean", torch.tensor([0.485, 0.456, 0.406]).view(1, 3, 1, 1))
+        self.register_buffer("rgb_std", torch.tensor([0.229, 0.224, 0.225]).view(1, 3, 1, 1))
+
+    def train(self, mode=True):
+        super().train(mode)
+        self.edge.eval()
+        for module in self.modules():
+            if isinstance(module, nn.BatchNorm2d):
+                module.eval()
+        return self
+
+    def forward(self, images):
+        with torch.no_grad():
+            normalized = (images - self.rgb_mean) / self.rgb_std
+            features = self.edge(normalized)
+            transmitted = self.blur(features)
+        return self.cloud(transmitted)

+def legacy_read_lock(path):
+    """제품에서 쓰지 않는 삭제 공유 없는 Windows 핸들을 검사 목적으로 엽니다."""
+    import ctypes
+    from ctypes import wintypes
+    kernel = ctypes.WinDLL("kernel32", use_last_error=True)
+    create_file = kernel.CreateFileW
+    create_file.argtypes = [wintypes.LPCWSTR, wintypes.DWORD, wintypes.DWORD, ctypes.c_void_p,
+                            wintypes.DWORD, wintypes.DWORD, wintypes.HANDLE]
+    create_file.restype = wintypes.HANDLE
+    close_handle = kernel.CloseHandle
+    close_handle.argtypes = [wintypes.HANDLE]
+    close_handle.restype = wintypes.BOOL
+    handle = create_file(str(path), 0x80000000, 3, None, 3, 128, None)
+    if handle == ctypes.c_void_p(-1).value:
+        raise ctypes.WinError(ctypes.get_last_error())
+    return lambda: close_handle(handle)
