+    "def focal_loss(gamma=2.0, label_smoothing=0.05):\n",
+    "    def loss_fn(y_true, y_pred):\n",
+    "        num_cls   = tf.cast(tf.shape(y_true)[-1], tf.float32)\n",
+    "        y_true_sm = y_true * (1.0 - label_smoothing) + (label_smoothing / num_cls)\n",
+    "        y_pred    = tf.clip_by_value(y_pred, 1e-7, 1.0)\n",
+    "        ce        = -y_true_sm * tf.math.log(y_pred)\n",
+    "        weight    = y_true_sm * tf.math.pow(1.0 - y_pred, gamma)\n",
+    "        return tf.reduce_mean(tf.reduce_sum(weight * ce, axis=-1))\n",
+    "    return loss_fn\n",

+    "def _gaussian_kernel_2d(sigma, max_radius):\n",
+    "    # 반환값: (kernel_2d, radius) 또는 완전 붕괴가 필요하면 None.\n",
+    "    radius_wanted = int(math.ceil(3 * sigma))\n",
+    "    if radius_wanted >= max_radius:\n",
+    "        return None\n",
+    "    ax = np.arange(-radius_wanted, radius_wanted + 1, dtype=np.float32)\n",
+    "    kernel_1d = np.exp(-(ax ** 2) / (2 * sigma ** 2))\n",
+    "    kernel_1d /= kernel_1d.sum()\n",
+    "    kernel_2d = np.outer(kernel_1d, kernel_1d)\n",
+    "    kernel_2d /= kernel_2d.sum()\n",
