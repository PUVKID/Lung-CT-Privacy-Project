+        "    img = img.astype(np.float64)\n",

+        "    \"\"\"단일 스케일에서 SSIM의 luminance(l) 성분과 contrast*structure(cs) 성분을 분리 계산.\"\"\"\n",

+        "    win_size = max(win_size, 3)\n",
